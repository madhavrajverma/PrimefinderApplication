// Test 2A — File Server Fault Tolerance: Server Crash During Read
//
// What this tests:
//   A worker opens and caches a large input file from AFS.
//   The primary AFS server (s1) is killed mid-read.
//   Because AFS uses whole-file caching, the read completes from local cache.
//   After restarting s1, a new client retries and reads successfully.
//
// How to run (local binary mode):
//   Prerequisites: s1, s2, s3 running. At least one worker running.
//
//   go run tests/test2a/main.go \
//     -servers localhost:50051,localhost:50052,localhost:50053 \
//     -cacheDir /tmp/test2a-cache
//
// How to run (Docker mode):
//   go run tests/test2a/main.go \
//     -servers localhost:50051,localhost:50052,localhost:50053 \
//     -cacheDir /tmp/test2a-cache \
//     -docker=true
//
// What PASS looks like:
//   [Step 1] File cached locally            PASS
//   [Step 2] s1 killed
//   [Step 3] Read completes from cache      PASS
//   [Step 4] New client finds new primary   PASS
//   [Step 5] Write on new primary succeeds  PASS
//   [Step 6] s1 restarted and synced        PASS

package main

import (
	"flag"
	"fmt"
	"io"
	"log"
	"os/exec"
	"strings"
	"time"

	"afsfs/pkg/afs"
)

func main() {
	servers  := flag.String("servers", "localhost:50051,localhost:50052,localhost:50053", "AFS server addresses")
	cacheDir := flag.String("cacheDir", "/tmp/test2a-cache", "local cache dir")
	docker   := flag.Bool("docker", false, "use docker kill/start instead of pkill")
	flag.Parse()

	fmt.Println("TEST 2A — Server Crash During Read")
	fmt.Println("Proves: whole-file cache survives primary crash; new client finds new primary")

	addrs := strings.Split(*servers, ",")

	// ── Step 1: Open file — triggers FetchFile, file lands in local cache ──
	fmt.Println("\n[Step 1] Connect and open input file (all servers alive)")
	client, err := afs.NewClient(addrs, *cacheDir)
	if err != nil {
		log.Fatalf("FAIL: cannot connect: %v", err)
	}

	h, err := client.Open("input_dataset_001.txt", false)
	if err != nil {
		log.Fatalf("FAIL: Open: %v", err)
	}
	fmt.Printf("  PASS ✓  opened handle=%d — file is now in local cache at %s\n", h, *cacheDir)

	// ── Step 2: Kill the primary (s1) ───────────────────────────────────
	fmt.Println("\n[Step 2] Killing primary server s1...")
	if *docker {
		if err := dockerKill("s1"); err != nil {
			fmt.Printf("  Warning: docker kill s1 failed: %v\n", err)
			fmt.Println("  Run manually: docker compose kill s1")
			fmt.Println("  Press ENTER when done...")
			fmt.Scanln()
		} else {
			fmt.Println("  s1 killed via docker. Waiting 5s for election...")
			time.Sleep(5 * time.Second)
		}
	} else {
		if err := pkillServer(50051); err != nil {
			fmt.Printf("  Warning: pkill failed: %v\n", err)
			fmt.Println("  Kill s1 manually, then press ENTER...")
			fmt.Scanln()
		} else {
			fmt.Println("  s1 killed. Waiting 5s for bully election to complete...")
			time.Sleep(5 * time.Second)
		}
	}

	// ── Step 3: Read from local cache — s1 is dead, no network needed ───
	fmt.Println("\n[Step 3] Read file while s1 is dead (reads from LOCAL CACHE)")
	buf := make([]byte, 64*1024)
	var data []byte
	for {
		n, e := client.Read(h, buf)
		if n > 0 {
			data = append(data, buf[:n]...)
		}
		if e == io.EOF || e != nil {
			break
		}
	}
	client.Close(h)

	if len(data) > 0 {
		fmt.Printf("  PASS ✓  read %d bytes from LOCAL CACHE — no network required\n", len(data))
		fmt.Printf("  First 80 chars: %.80s\n", string(data))
	} else {
		fmt.Println("  FAIL: nothing read from cache")
	}
	client.CloseConn()

	// ── Step 4: New client — must discover new primary automatically ──────
	fmt.Println("\n[Step 4] New client connects — must find new primary (s2 or s3)")
	client2, err := afs.NewClient(addrs, *cacheDir+"-verify")
	if err != nil {
		fmt.Printf("  FAIL: could not find new primary: %v\n", err)
		fmt.Println("  Is at least one of s2/s3 alive? Check: docker compose ps")
		return
	}
	defer client2.CloseConn()
	fmt.Println("  PASS ✓  connected to new primary (bully election succeeded)")

	// ── Step 5: Write to new primary — proves it accepts writes ──────────
	fmt.Println("\n[Step 5] Write to new primary — proves election winner is functional")
	payload := "written after s1 crash — new primary works\n"
	client2.DeleteFile("test2a_after_crash.txt")
	wh, err := client2.CreateFile("test2a_after_crash.txt")
	if err != nil {
		fmt.Printf("  FAIL: CreateFile on new primary: %v\n", err)
		return
	}
	client2.Write(wh, []byte(payload))
	if err := client2.Close(wh); err != nil {
		fmt.Printf("  FAIL: Close (StoreFile): %v\n", err)
		return
	}
	fmt.Println("  PASS ✓  write succeeded on new primary and replicated to surviving replica")

	// ── Step 6: Restart s1 and verify it syncs ───────────────────────────
	fmt.Println("\n[Step 6] Restart s1 — it should sync missed files from new primary")
	if *docker {
		if err := dockerStart("s1"); err != nil {
			fmt.Printf("  Warning: docker start s1 failed: %v\n", err)
			fmt.Println("  Restart manually: docker compose start s1")
			fmt.Println("  Press ENTER when s1 is back...")
			fmt.Scanln()
		} else {
			fmt.Println("  s1 restarted. Waiting 8s for SyncState to complete...")
			time.Sleep(8 * time.Second)
		}
	} else {
		fmt.Println("  Restart s1 manually in another terminal:")
		fmt.Println("  cd afsfs && ./bin/server -id s1 -host localhost -port 50051 \\")
		fmt.Println("    -primary=false -peers s2=localhost:50052,s3=localhost:50053 \\")
		fmt.Println("    -inputDir ./testdata/input -outputDir ./testdata/output-s1 &")
		fmt.Println("  Press ENTER when s1 is back...")
		fmt.Scanln()
	}

	// Verify s1 now has the file written while it was dead
	client3, err := afs.NewClient([]string{addrs[0]}, *cacheDir+"-s1check")
	if err != nil {
		fmt.Printf("  FAIL: cannot connect to s1 after restart: %v\n", err)
		return
	}
	defer client3.CloseConn()

	rh, err := client3.Open("test2a_after_crash.txt", false)
	if err != nil {
		fmt.Printf("  FAIL: s1 does not have test2a_after_crash.txt after sync: %v\n", err)
		return
	}
	var synced []byte
	for {
		n, e := client3.Read(rh, buf)
		if n > 0 {
			synced = append(synced, buf[:n]...)
		}
		if e == io.EOF || e != nil {
			break
		}
	}
	client3.Close(rh)

	if string(synced) == payload {
		fmt.Println("  PASS ✓  s1 has synced the file written while it was dead")
	} else {
		fmt.Printf("  FAIL: s1 content wrong — got %q\n", string(synced))
	}

	fmt.Println("\nTEST 2A COMPLETE")
	fmt.Println("\nVerify manually:")
	fmt.Println("  cat afsfs/testdata/output-s1/test2a_after_crash.txt")
	fmt.Println("  cat afsfs/testdata/output-s2/test2a_after_crash.txt")
}

func pkillServer(port int) error {
	return exec.Command("sh", "-c",
		fmt.Sprintf("pkill -f 'server.*%d' || pkill -f '%d'", port, port)).Run()
}

func dockerKill(container string) error {
	return exec.Command("docker", "compose", "kill", container).Run()
}

func dockerStart(container string) error {
	return exec.Command("docker", "compose", "start", container).Run()
}
