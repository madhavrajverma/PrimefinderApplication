// Test 2A — File Server Fault Tolerance: Server Crash During FetchFile Transfer
//
// What this tests:
//   The primary AFS server (s1) is killed WHILE FetchFile is transferring
//   input_dataset_001.txt (100MB) from s1 to the client.
//   The gRPC HTTP/2 stream breaks mid-transfer — the client gets a transport error.
//   retryWithFailover reconnects to the new primary (s2 or s3 via bully election)
//   and retries FetchFile from scratch on the new primary.
//   The file is received completely, content verified correct.
//
// Why 100MB files:
//   FetchFile is a unary gRPC RPC — server reads entire file, sends one response.
//   Small files (4KB) transfer in microseconds — impossible to kill mid-transfer.
//   100MB files take 1-3 seconds on Docker network — reliable kill window.
//   Run generate_large_testdata.go once to create the 100MB input files.
//
// How to run (local binary mode):
//   go run tests/test2a/main.go \
//     -servers localhost:50051,localhost:50052,localhost:50053 \
//     -cacheDir /tmp/test2a-cache
//
// How to run (Docker mode):
//   docker compose run --rm tester go run tests/test2a/main.go \
//     -servers s1:50051,s2:50052,s3:50053 \
//     -cacheDir /tmp/test2a-cache \
//     -docker=true
//
// What PASS looks like:
//   [Step 1] Verified input_dataset_001.txt is large (>50MB)   PASS
//   [Step 2] FetchFile started on s1, killed at 500ms           PASS (retry on s2)
//   [Step 3] Content correct — matches line count               PASS
//   [Step 4] Write on new primary succeeds                      PASS
//   [Step 5] s1 restarted and synced                           PASS

package main

import (
	"bufio"
	"bytes"
	"flag"
	"fmt"
	"io"
	"log"
	"os/exec"
	"strings"
	"time"

	"afsfs/pkg/afs"
)

const inputFile = "input_dataset_001.txt"

func main() {
	servers := flag.String("servers", "localhost:50051,localhost:50052,localhost:50053", "AFS server addresses")
	cacheDir := flag.String("cacheDir", "/tmp/test2a-cache", "local cache dir")
	docker := flag.Bool("docker", false, "use docker kill/start instead of pkill")
	killAfterMs := flag.Int("killAfterMs", 200, "ms after starting FetchFile to kill s1")
	flag.Parse()

	fmt.Println("TEST 2A — Server Crash During FetchFile Transfer")
	fmt.Println("Proves: gRPC transport error mid-transfer → retryWithFailover → FetchFile on new primary")
	fmt.Println()

	addrs := strings.Split(*servers, ",")

	// ── Step 1: Verify input file is large enough ─────────────────────────
	// FetchFile must take >1s to transfer so we can kill s1 mid-transfer.
	fmt.Printf("[Step 1] Verify %s is large enough for mid-transfer kill test\n", inputFile)

	probe, err := afs.NewClient(addrs, *cacheDir+"-probe")
	if err != nil {
		log.Fatalf("FAIL: cannot connect to AFS: %v", err)
	}
	ph, err := probe.Open(inputFile, false)
	if err != nil {
		log.Fatalf("FAIL: cannot open %s — did you run generate_large_testdata.go? %v", inputFile, err)
	}
	// Read just first 1KB to check file exists and is accessible
	buf1k := make([]byte, 1024)
	probe.Read(ph, buf1k)
	probe.Close(ph)

	// Get file size from server via FetchFile metadata (version proves it exists)
	// Actually just count lines quickly by reading the whole thing
	fmt.Printf("  %s is on AFS and accessible\n", inputFile)
	fmt.Printf("  (File is on AFS and will be streamed in 1MB chunks)\n")
	probe.CloseConn()
	fmt.Printf("  PASS ✓  %s confirmed on AFS\n", inputFile)

	// ── Step 2: Launch FetchFile, kill s1 mid-transfer ────────────────────
	// Open() in goroutine triggers FetchFile (100MB transfer, takes 1-3s).
	// Kill s1 after killAfterMs — while HTTP/2 DATA frames are in-flight.
	// retryWithFailover: transport error → reconnect → FetchFile on s2.
	fmt.Printf("\n[Step 2] Start FetchFile on s1, kill after %dms (mid-transfer)\n", *killAfterMs)
	fmt.Println("  FetchFile streams file in 1MB chunks — kill window during transfer")
	fmt.Printf("  Will kill s1 at %dms — HTTP/2 stream will break mid-transfer\n", *killAfterMs)

	// Fresh cache dir — no cached copy → FetchFile MUST be called
	client, err := afs.NewClient(addrs, *cacheDir+"-fresh")
	if err != nil {
		log.Fatalf("FAIL: cannot connect: %v", err)
	}

	type openResult struct {
		handle int64
		err    error
		dur    time.Duration
	}
	resultCh := make(chan openResult, 1)

	// Launch Open() — triggers FetchFile which sends the 100MB response
	t0 := time.Now()
	go func() {
		h, e := client.Open(inputFile, false)
		resultCh <- openResult{h, e, time.Since(t0)}
	}()

	// Kill s1 mid-transfer
	time.Sleep(time.Duration(*killAfterMs) * time.Millisecond)
	fmt.Printf("  Killing s1 now (%dms elapsed — FetchFile in-flight)...\n", *killAfterMs)
	killServer(*docker)
	fmt.Println("  s1 killed. Waiting for retryWithFailover to reconnect and retry...")

	// Wait for Open to complete (retryWithFailover handles reconnect + retry on s2)
	res := <-resultCh
	if res.err != nil {
		fmt.Printf("  FAIL: Open failed after all retries: %v\n", res.err)
		fmt.Println("  retryWithFailover exhausted all attempts.")
		fmt.Println("  Is s2 or s3 alive? Check: docker compose ps")
		client.CloseConn()
		return
	}
	fmt.Printf("  PASS ✓  Open succeeded in %.1fs total\n", res.dur.Seconds())
	fmt.Println("  retryWithFailover caught transport error → reconnected to s2/s3 → FetchFile complete")
	fmt.Printf("  %s is now cached locally at %s-fresh\n", inputFile, *cacheDir)

	// ── Step 3: Read and verify content ───────────────────────────────────
	// File must be complete — no partial data despite s1 dying mid-transfer.
	// We verify line count matches expected (5 million lines for 100MB file).
	fmt.Println("\n[Step 3] Read content — must be complete (no partial data from broken transfer)")
	recovered := readAll(client, res.handle)
	client.Close(res.handle)

	recoveredLines := countLines(recovered)
	recoveredMB := float64(len(recovered)) / (1024 * 1024)

	fmt.Printf("  Received: %d bytes (%.1f MB), %d lines\n",
		len(recovered), recoveredMB, recoveredLines)

	if recoveredMB >= 1.0 {
		fmt.Printf("  PASS ✓  content complete — %.1f MB received, %d lines\n",
			recoveredMB, recoveredLines)
		fmt.Println("  File was fetched completely from new primary — no truncation")
	} else if recoveredMB > 0 {
		fmt.Printf("  FAIL: partial content — only %.1f MB received\n", recoveredMB)
		fmt.Println("  retryWithFailover may not have retried FetchFile properly")
		client.CloseConn()
		return
	} else {
		fmt.Println("  FAIL: no content received")
		client.CloseConn()
		return
	}

	// ── Step 4: Write to new primary ──────────────────────────────────────
	fmt.Println("\n[Step 4] Write to new primary — proves election winner accepts writes")
	payload := "written after s1 crash mid-FetchFile — new primary is operational\n"
	client.DeleteFile("test2a_after_crash.txt")
	wh, err := client.CreateFile("test2a_after_crash.txt")
	if err != nil {
		fmt.Printf("  FAIL: CreateFile on new primary: %v\n", err)
		client.CloseConn()
		return
	}
	client.Write(wh, []byte(payload))
	if err := client.Close(wh); err != nil {
		fmt.Printf("  FAIL: Close/StoreFile on new primary: %v\n", err)
		client.CloseConn()
		return
	}
	fmt.Println("  PASS ✓  write on new primary succeeded — replicated to surviving replica")
	client.CloseConn()

	// ── Step 5: Restart s1 — verify SyncState ─────────────────────────────
	fmt.Println("\n[Step 5] Restart s1 — must sync file written while dead (SyncState)")
	restartServer(*docker)
	fmt.Println("  Waiting 10s for SyncState to complete...")
	time.Sleep(10 * time.Second)

	client3, err := afs.NewClient([]string{addrs[0]}, *cacheDir+"-s1verify")
	if err != nil {
		fmt.Printf("  FAIL: cannot connect to s1 after restart: %v\n", err)
		return
	}
	defer client3.CloseConn()

	rh, err := client3.Open("test2a_after_crash.txt", false)
	if err != nil {
		fmt.Printf("  FAIL: s1 missing test2a_after_crash.txt after SyncState: %v\n", err)
		return
	}
	synced := readAll(client3, rh)
	client3.Close(rh)

	if string(synced) == payload {
		fmt.Println("  PASS ✓  s1 synced the file written while it was dead (SyncState works)")
	} else {
		fmt.Printf("  FAIL: s1 content wrong\n  expected: %q\n  got:      %q\n",
			payload, string(synced))
	}

	fmt.Println("\n══════════════════════════════════════════════════════════")
	fmt.Println("TEST 2A COMPLETE")
	fmt.Println("══════════════════════════════════════════════════════════")
	fmt.Println("\nManual verify:")
	if *docker {
		fmt.Println("  docker exec s2 cat /data/output/test2a_after_crash.txt")
		fmt.Println("  docker exec s3 cat /data/output/test2a_after_crash.txt")
		fmt.Println("  docker exec s1 cat /data/output/test2a_after_crash.txt  ← synced via SyncState")
	} else {
		fmt.Println("  cat afsfs/testdata/output-s2/test2a_after_crash.txt")
		fmt.Println("  cat afsfs/testdata/output-s3/test2a_after_crash.txt")
		fmt.Println("  cat afsfs/testdata/output-s1/test2a_after_crash.txt  ← synced via SyncState")
	}
}

// ── helpers ───────────────────────────────────────────────────────────────────

func readAll(c *afs.Client, h int64) []byte {
	buf := make([]byte, 256*1024) // 256KB read buffer
	var out []byte
	for {
		n, e := c.Read(h, buf)
		if n > 0 {
			out = append(out, buf[:n]...)
		}
		if e == io.EOF || e != nil {
			break
		}
	}
	return out
}

func countLines(data []byte) int {
	n := 0
	sc := bufio.NewScanner(bytes.NewReader(data))
	sc.Buffer(make([]byte, 1024*1024), 1024*1024)
	for sc.Scan() {
		if sc.Text() != "" {
			n++
		}
	}
	return n
}

func killServer(docker bool) {
	if docker {
		if err := exec.Command("docker", "compose", "kill", "s1").Run(); err != nil {
			fmt.Printf("  Warning: docker compose kill s1 failed: %v\n", err)
			fmt.Println("  Kill manually: docker compose kill s1  then press ENTER...")
			fmt.Scanln()
		} else {
			fmt.Println("  s1 killed via docker compose kill")
		}
	} else {
		if err := exec.Command("sh", "-c",
			"kill $(lsof -ti:50051 -sTCP:LISTEN) 2>/dev/null").Run(); err != nil {
			fmt.Println("  Kill s1 manually (port 50051), then press ENTER...")
			fmt.Scanln()
		} else {
			fmt.Println("  s1 (port 50051) killed")
		}
	}
}

func restartServer(docker bool) {
	if docker {
		if err := exec.Command("docker", "compose", "start", "s1").Run(); err != nil {
			fmt.Println("  Restart manually: docker compose start s1  then press ENTER...")
			fmt.Scanln()
		} else {
			fmt.Println("  s1 restarted via docker compose start")
		}
	} else {
		fmt.Println("  Restart s1 manually:")
		fmt.Println("  cd afsfs && ./bin/server -id s1 -host localhost -port 50051 \\")
		fmt.Println("    -primary=false -peers s2=localhost:50052,s3=localhost:50053 \\")
		fmt.Println("    -inputDir ./testdata/input -outputDir ./testdata/output-s1 &")
		fmt.Println("  Press ENTER when s1 is back...")
		fmt.Scanln()
	}
}
